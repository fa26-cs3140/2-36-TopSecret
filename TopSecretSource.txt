// Rules for user accounts (Member D): validation, first account,
// adding users, logging in, and setting or changing passwords.
public class AccountManager {

    public static final String USERNAME_RULE =
            "Usernames must be at least 3 characters and use only letters, numbers, - and _.";
    public static final String PASSWORD_RULE =
            "Passwords must be at least 5 characters and include uppercase and lowercase letters.";

    private final UserStore store;

    public AccountManager(UserStore store) {
        this.store = store;
    }

    // At least 3 characters: letters, numbers, hyphens, underscores
    public static boolean isValidUsername(String username) {
        return username != null && username.matches("[A-Za-z0-9_-]{3,}");
    }

    // At least 5 characters with an uppercase and a lowercase letter.
    // Tabs and line breaks are not allowed because of the file format.
    public static boolean isValidPassword(String password) {
        return password != null
                && password.length() >= 5
                && !password.equals(password.toLowerCase())
                && !password.equals(password.toUpperCase())
                && !password.contains("\t")
                && !password.contains("\n")
                && !password.contains("\r");
    }

    public boolean hasAccounts() {
        return !store.loadAll().isEmpty();
    }

    public List<UserAccount> getUsers() {
        return store.loadAll();
    }

    // The account with this username, or null
    public UserAccount findUser(String username) {
        for (UserAccount u : store.loadAll()) {
            if (u.getUsername().equals(username)) {
                return u;
            }
        }
        return null;
    }

    // First account in an empty system. It is an admin.
    public UserAccount createFirstAccount(String username, String password) {
        if (hasAccounts()) {
            throw new IllegalStateException("Accounts already exist");
        }
        requireValid(username, password);
        UserAccount first = new UserAccount(username, password, true);
        store.saveAll(List.of(first));
        return first;
    }

    // A logged-in user adds a new user by username only.
    // The new user sets their own password at first login.
    public UserAccount addUser(UserAccount creator, String newUsername) {
        if (creator == null) {
            throw new IllegalStateException("You must be logged in to add a user");
        }
        if (!isValidUsername(newUsername)) {
            throw new IllegalArgumentException(USERNAME_RULE);
        }
        List<UserAccount> users = new ArrayList<>(store.loadAll());
        for (UserAccount u : users) {
            if (u.getUsername().equals(newUsername)) {
                throw new IllegalArgumentException("Username already exists");
            }
        }
        UserAccount added = new UserAccount(newUsername, null, false);
        users.add(added);
        store.saveAll(users);
        return added;
    }

    // The user if the username and password match, otherwise null
    public UserAccount login(String username, String password) {
        UserAccount user = findUser(username);
        if (user == null || !user.hasPassword() || !user.getPassword().equals(password)) {
            return null;
        }
        return user;
    }

    // True for a user who was added but has not set a password yet
    public boolean needsPassword(String username) {
        UserAccount user = findUser(username);
        return user != null && !user.hasPassword();
    }

    // First-login password for a new user
    public boolean setInitialPassword(String username, String password, String confirm) {
        if (!needsPassword(username) || !isValidPassword(password) || !password.equals(confirm)) {
            return false;
        }
        return savePassword(username, password);
    }

    // Change a password after checking the current one
    public boolean changePassword(String username, String oldPassword,
                                  String newPassword, String confirm) {
        if (login(username, oldPassword) == null
                || !isValidPassword(newPassword) || !newPassword.equals(confirm)) {
            return false;
        }
        return savePassword(username, newPassword);
    }

    private boolean savePassword(String username, String password) {
        List<UserAccount> users = new ArrayList<>(store.loadAll());
        for (UserAccount u : users) {
            if (u.getUsername().equals(username)) {
                u.setPassword(password);
                store.saveAll(users);
                return true;
            }
        }
        return false;
    }

    private static void requireValid(String username, String password) {
        if (!isValidUsername(username)) {
            throw new IllegalArgumentException(USERNAME_RULE);
        }
        if (!isValidPassword(password)) {
            throw new IllegalArgumentException(PASSWORD_RULE);
        }
    }
}

import java.util.Map;

public class Cipher {
    // maps each ciphered character back to its real character
    private final Map<Character, Character> map = new HashMap<>();
    // maps each real character to its ciphered character
    private final Map<Character, Character> reverse = new HashMap<>();

    public Cipher(String key) {
        if (key == null || key.isBlank()) {
            return; // empty key: text is left unchanged
        }
        String[] lines = key.strip().split("\\R");
        if (lines.length != 2 || lines[0].length() != lines[1].length()) {
            throw new IllegalArgumentException("Invalid key: must be 2 lines of the same length");
        }
        for (int i = 0; i < lines[0].length(); i++) {
            map.put(lines[1].charAt(i), lines[0].charAt(i));
            reverse.put(lines[0].charAt(i), lines[1].charAt(i));
        }
    }

    public String decipher(String text) {
        StringBuilder result = new StringBuilder();
        for (char c : text.toCharArray()) {
            result.append(map.getOrDefault(c, c));
        }
        return result.toString();
    }

    // The opposite of decipher: turns real text into ciphered text
    public String encipher(String text) {
        StringBuilder result = new StringBuilder();
        for (char c : text.toCharArray()) {
            result.append(reverse.getOrDefault(c, c));
        }
        return result.toString();
    }
}

// Saves user accounts in a ciphered file (Member D).
// Each line before ciphering is: username <tab> password <tab> ADMIN or USER
// An empty password means the user has not set one yet.
public class CipheredUserStore implements UserStore {

    public static final Path DEFAULT_USERS_FILE = Path.of("users.cip");
    public static final Path DEFAULT_KEY_FILE = Path.of("ciphers", "key.txt");

    // Used when ciphers/key.txt is missing, so the program still runs
    static final String BUILT_IN_KEY =
            "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ1234567890\n"
          + "bcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ1234567890a";

    private final Path usersFile;
    private final Cipher cipher;

    public CipheredUserStore() {
        this(DEFAULT_USERS_FILE, DEFAULT_KEY_FILE);
    }

    public CipheredUserStore(Path usersFile, Path keyFile) {
        this.usersFile = usersFile;
        this.cipher = new Cipher(readKey(keyFile));
    }

    private static String readKey(Path keyFile) {
        try {
            return Files.exists(keyFile) ? Files.readString(keyFile) : BUILT_IN_KEY;
        } catch (IOException e) {
            throw new UncheckedIOException("Cannot read cipher key", e);
        }
    }

    @Override
    public List<UserAccount> loadAll() {
        List<UserAccount> users = new ArrayList<>();
        if (!Files.exists(usersFile)) {
            return users;
        }
        try {
            String text = cipher.decipher(Files.readString(usersFile));
            for (String line : text.split("\\R")) {
                String[] parts = line.split("\t", -1);
                if (parts.length == 3 && !parts[0].isEmpty()) {
                    String password = parts[1].isEmpty() ? null : parts[1];
                    users.add(new UserAccount(parts[0], password, parts[2].equals("ADMIN")));
                }
            }
        } catch (IOException e) {
            throw new UncheckedIOException("Cannot read user accounts", e);
        }
        return users;
    }

    @Override
    public void saveAll(List<UserAccount> users) {
        StringBuilder text = new StringBuilder();
        for (UserAccount u : users) {
            text.append(u.getUsername()).append('\t')
                .append(u.hasPassword() ? u.getPassword() : "").append('\t')
                .append(u.isAdmin() ? "ADMIN" : "USER").append('\n');
        }
        try {
            Path parent = usersFile.toAbsolutePath().getParent();
            Files.createDirectories(parent);
            Files.writeString(usersFile, cipher.encipher(text.toString()));
        } catch (IOException e) {
            throw new UncheckedIOException("Cannot save user accounts", e);
        }
    }
}

public class FileHandler {

    private final Path dataFolder = Path.of("data");
    private final Path cipherFolder = Path.of("ciphers");

    public String readDataFile(String fileName) { // Reads a mission file from the data folder and return its contents
        try {
            return Files.readString(dataFolder.resolve(fileName));
        } catch (Exception e) {
            return null; // Return null if the file can not be found or the name is invalid
        }
    }

    public String readCipherFile(String fileName) {    // Reads a cipher key from the ciphers folder
        try {
            return Files.readString(cipherFolder.resolve(fileName));
        } catch (Exception e) {
            return null; // Return null if the file can not be found or the name is invalid
        }
    }

    public List<String> getAvailableFiles() {   // Returns a list of available mission files in the data folder

        List<String> files = new ArrayList<>(); // Stores the available file names

        try {
            Files.list(dataFolder)
                    .filter(Files::isRegularFile)  // Only includes regular files
                    .forEach(file -> // Adds each file name to the list
                            files.add(file.getFileName().toString()));

        } catch (IOException e) {
            return files; // Returns current list
        }

        Collections.sort(files); //Sorts the files names

        return files;
    }

// Console prompts for accounts (Member D, built on Member B's Homework 4 login):
// first-run setup, login, first-login password, and changing a password.
public class LoginHandler {

    private final AccountManager accounts;
    private final Scanner in;
    private final PrintStream out;

    public LoginHandler(AccountManager accounts, Scanner in, PrintStream out) {
        this.accounts = accounts;
        this.in = in;
        this.out = out;
    }

    // Returns the logged-in user, or null if login did not succeed.
    // With no accounts yet, creates the first (admin) account and returns null
    // so the program exits, as in Homework 4.
    public UserAccount authenticate() {
        if (!accounts.hasAccounts()) {
            createFirstAccount();
            return null;
        }

        String username = ask("Username: ");
        if (username == null) {
            return null;
        }

        if (accounts.needsPassword(username)) {
            return setFirstPassword(username);
        }

        String password = ask("Password: ");
        UserAccount user = password == null ? null : accounts.login(username, password);
        if (user == null) {
            out.println("Incorrect username or password.");
            return null;
        }
        out.println("Login successful!");
        return user;
    }

    // For: java TopSecret --change-password
    public boolean changePassword() {
        String username = ask("Username: ");
        String oldPassword = ask("Current password: ");
        String newPassword = ask("New password: ");
        String confirm = ask("Confirm new password: ");
        if (confirm != null && accounts.changePassword(username, oldPassword, newPassword, confirm)) {
            out.println("Password changed successfully.");
            return true;
        }
        out.println("Password change failed. " + AccountManager.PASSWORD_RULE);
        return false;
    }

    private void createFirstAccount() {
        out.println("No user accounts found. Create the first account (it will be an admin).");
        out.println(AccountManager.USERNAME_RULE);
        out.println(AccountManager.PASSWORD_RULE);
        String username = ask("New username: ");
        String password = ask("New password: ");
        String confirm = ask("Confirm password: ");
        if (confirm == null) {
            return;
        }
        if (!password.equals(confirm)) {
            out.println("Passwords do not match.");
            return;
        }
        try {
            accounts.createFirstAccount(username, password);
            out.println("Account created. Restart the program to log in.");
        } catch (IllegalArgumentException e) {
            out.println(e.getMessage());
        }
    }

    private UserAccount setFirstPassword(String username) {
        out.println("Welcome, " + username + ". Create your password.");
        out.println(AccountManager.PASSWORD_RULE);
        String password = ask("New password: ");
        String confirm = ask("Confirm password: ");
        if (confirm != null && accounts.setInitialPassword(username, password, confirm)) {
            out.println("Password set. Login successful!");
            return accounts.findUser(username);
        }
        out.println("Password not set. Passwords must match and follow the rules.");
        return null;
    }

    // Prints the prompt and reads a line, or returns null if there is no more input
    private String ask(String prompt) {
        out.print(prompt);
        return in.hasNextLine() ? in.nextLine() : null;
    }
}

// Mission record retrieved from the SQLite database.
public class Mission {
    private final int id;
    private final String title;
    private final String brief;
    private final String date;
    private final int facilityId;
    private final List<Integer> agentIds;

    // Original Homework 4 constructor
    public Mission(int id, String title,
                   String brief, String date) {
        this.id = id;
        this.title = title;
        this.brief = brief;
        this.date = date;
        this.facilityId = 0;
        this.agentIds = new ArrayList<>();
    }

    // Homework 5 constructor
    public Mission(int id, String title,
                   String brief, String date,
                   int facilityId,
                   List<Integer> agentIds) {

        if (facilityId <= 0) {
            throw new IllegalArgumentException(
                    "Mission must have a facility");
        }

        if (agentIds == null || agentIds.isEmpty()
                || agentIds.contains(null)) {
            throw new IllegalArgumentException(
                    "Mission must have at least one agent");
        }

        this.id = id;
        this.title = title;
        this.brief = brief;
        this.date = date;
        this.facilityId = facilityId;
        this.agentIds = new ArrayList<>(agentIds);
    }

    // Getter methods
    public int getId() {
        return id;
    }

    public String getTitle() {
        return title;
    }

    public String getBrief() {
        return brief;
    }

    public String getDate() {
        return date;
    }

    public int getFacilityId() {
        return facilityId;
    }

    public List<Integer> getAgentIds() {
        return new ArrayList<>(agentIds);
    }

    // Equals method from teammate's code
    @Override
    public boolean equals(Object o) {
        if (this == o) {
            return true;
        }

        if (!(o instanceof Mission)) {
            return false;
        }

        Mission other = (Mission) o;

        return id == other.id
                && Objects.equals(title, other.title)
                && Objects.equals(brief, other.brief)
                && Objects.equals(date, other.date);
    }

    // HashCode method from teammate's code
    @Override
    public int hashCode() {
        return Objects.hash(id, title, brief, date);
    }
}

// Interactive menu shown after the user logs in (Member D).
// Keeps running until the user chooses Exit.
public class MissionMenu {

    private final MissionRepository missions;
    private final MissionSearch search;
    private final Scanner in;
    private final PrintStream out;
    private final AuditLog audit;
    private final AccountManager accounts;
    private final UserAccount user;

    // Homework 4 menu: no audit log and no account options
    public MissionMenu(MissionRepository missions, MissionSearch search, Scanner in, PrintStream out) {
        this(missions, search, in, out, null, null, null);
    }

    // Homework 5 menu: records reads and creates in the audit log for the logged-in user
    public MissionMenu(MissionRepository missions, MissionSearch search, Scanner in, PrintStream out,
                       AuditLog audit, AccountManager accounts, UserAccount user) {
        this.missions = missions;
        this.search = search;
        this.in = in;
        this.out = out;
        this.audit = audit;
        this.accounts = accounts;
        this.user = user;
    }

    public void run() {
        while (true) {
            showMenu();
            if (!in.hasNextLine()) {
                return; // no more input
            }
            String choice = in.nextLine().trim();
            switch (choice) {
                case "1" -> listMissions();
                case "2" -> readMission();
                case "3" -> searchMissions();
                case "4" -> {
                    out.println("Goodbye.");
                    return;
                }
                case "5" -> {
                    if (accounts != null) {
                        addUser();
                    } else {
                        out.println("Invalid choice. Enter 1, 2, 3, or 4.");
                    }
                }
                default -> out.println("Invalid choice. Enter 1, 2, 3, or 4.");
            }
        }
    }

    void showMenu() {
        out.println();
        out.println("1. List missions");
        out.println("2. Read a mission");
        out.println("3. Search missions");
        out.println("4. Exit");
        if (accounts != null) {
            out.println("5. Add user");
        }
        out.print("Choose an option: ");
    }

    void listMissions() {
        List<Mission> all = missions.getAllMissions();
        if (all.isEmpty()) {
            out.println("No missions available.");
            return;
        }
        for (int i = 0; i < all.size(); i++) {
            out.println((i + 1) + ". " + all.get(i).getTitle());
        }
    }

    void readMission() {
        out.print("Enter mission number: ");
        if (!in.hasNextLine()) {
            return;
        }
        String input = in.nextLine().trim();
        List<Mission> all = missions.getAllMissions();
        int number;
        try {
            number = Integer.parseInt(input);
        } catch (NumberFormatException e) {
            out.println("Error: '" + input + "' is not a number.");
            return;
        }
        if (number < 1 || number > all.size()) {
            out.println("Error: No mission with number " + number + ".");
            return;
        }
        Mission m = all.get(number - 1);
        out.println(m.getTitle() + " (" + m.getDate() + ")");
        out.println(m.getBrief());
        logEvent(AuditLog.READ, "Mission brief #" + m.getId() + ": " + m.getTitle());
    }

    void addUser() {
        out.print("New username: ");
        if (!in.hasNextLine()) {
            return;
        }
        String name = in.nextLine().trim();
        try {
            accounts.addUser(user, name);
            out.println("User '" + name + "' added. They will create a password at first login.");
            logEvent(AuditLog.CREATE, "User account: " + name);
        } catch (IllegalArgumentException | IllegalStateException e) {
            out.println("Error: " + e.getMessage());
        }
    }

    // Records an event in the audit log, if this menu has one
    private void logEvent(String action, String item) {
        if (audit != null && user != null) {
            audit.record(user.getUsername(), action, item);
        }
    }

    void searchMissions() {
        out.print("Enter a word or phrase: ");
        if (!in.hasNextLine()) {
            return;
        }
        String query = in.nextLine();
        List<Mission> matches = search.searchBriefs(query);
        if (matches.isEmpty()) {
            out.println("No matches found.");
            return;
        }
        for (Mission m : matches) {
            out.println("- " + m.getTitle());
        }
    }

public interface MissionSearch {
    List<Mission> searchBriefs(String query);
}import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;

// This implements the search functionality and case-insensitive searches
public class MissionSearchService implements MissionSearch {
    private final Connection connection;

    public MissionSearchService(Connection connection) {
        this.connection = connection;
    }

    @Override
    public List<Mission> searchBriefs(String query) {
        List<Mission> results = new ArrayList<>();
// Return empty list immediately if the search query is null or blank
        if (query == null || query.trim().isEmpty()) {
            return results;
        }

        String sql = "SELECT id, title, brief, date FROM missions WHERE LOWER(brief) LIKE LOWER(?)";

        try (PreparedStatement stmt = connection.prepareStatement(sql)) {
            stmt.setString(1, "%" + query.trim() + "%");
            try (ResultSet rs = stmt.executeQuery()) {
                while (rs.next()) {
                    results.add(new Mission(
                            rs.getInt("id"),
                            rs.getString("title"),
                            rs.getString("brief"),
                            rs.getString("date")
                    ));
                }
            }
        } catch (SQLException e) {
            System.err.println("Search database query failed: " + e.getMessage());
        }

        return results;
    }

    public String formatSearchResults(String query, List<Mission> matches) {
        if (matches == null || matches.isEmpty()) {
            return "No matches found for: \"" + query + "\"";
        }

        StringBuilder sb = new StringBuilder();
        sb.append("Matches found (").append(matches.size()).append("):\n");
        for (Mission m : matches) {
            sb.append("[").append(m.getId()).append("] ").append(m.getTitle()).append("\n");
        }
        return sb.toString();
    }

public class ModelDatabase {
    private final Connection connection;

    public ModelDatabase(Connection connection) throws SQLException {
        this.connection = Objects.requireNonNull(connection);
        if (!connection.getAutoCommit())
            throw new SQLException("Connection must use auto-commit");

        try (Statement s = connection.createStatement()) {
            s.execute("PRAGMA foreign_keys = ON");
        }
    }

    public void createTables() throws SQLException {
        try (Statement s = connection.createStatement()) {
            s.executeUpdate("CREATE TABLE IF NOT EXISTS facilities (" +
                    "id INTEGER PRIMARY KEY, name TEXT NOT NULL, abbreviation TEXT NOT NULL)");

            s.executeUpdate("CREATE TABLE IF NOT EXISTS agents (" +
                    "id INTEGER PRIMARY KEY, name TEXT NOT NULL, " +
                    "date_of_birth TEXT, date_of_death TEXT, notes TEXT, " +
                    "facility_id INTEGER NOT NULL REFERENCES facilities(id) ON DELETE RESTRICT)");

            s.executeUpdate("CREATE TABLE IF NOT EXISTS briefs (" +
                    "id INTEGER PRIMARY KEY, title TEXT NOT NULL, " +
                    "date TEXT NOT NULL, text TEXT NOT NULL)");

            s.executeUpdate("CREATE TABLE IF NOT EXISTS mission_facilities (" +
                    "mission_id INTEGER PRIMARY KEY REFERENCES missions(id) ON DELETE CASCADE, " +
                    "facility_id INTEGER NOT NULL REFERENCES facilities(id) ON DELETE RESTRICT)");

            s.executeUpdate("CREATE TABLE IF NOT EXISTS mission_agents (" +
                    "mission_id INTEGER NOT NULL REFERENCES missions(id) ON DELETE CASCADE, " +
                    "agent_id INTEGER NOT NULL REFERENCES agents(id) ON DELETE RESTRICT, " +
                    "PRIMARY KEY(mission_id, agent_id))");
        }
    }

    private int execute(String sql, Object... values) throws SQLException {
        try (PreparedStatement p = connection.prepareStatement(sql)) {
            for (int i = 0; i < values.length; i++)
                p.setObject(i + 1, values[i]);
            return p.executeUpdate();
        }
    }

    private int getLastInsertId() throws SQLException {
        try (Statement s = connection.createStatement();
             ResultSet r = s.executeQuery("SELECT last_insert_rowid()")) {
            if (r.next()) return r.getInt(1);
        }
        throw new SQLException("No ID returned");
    }

    public int addFacility(String name, String abbreviation) throws SQLException {
        execute("INSERT INTO facilities(name, abbreviation) VALUES (?,?)",
                name, abbreviation);
        return getLastInsertId();
    }

    public int addAgent(String name, String dob, String dod,
                        String notes, int facilityId) throws SQLException {
        execute("INSERT INTO agents(name,date_of_birth,date_of_death,notes,facility_id) " +
                "VALUES (?,?,?,?,?)", name, dob, dod, notes, facilityId);
        return getLastInsertId();
    }

    public int addBrief(String title, String date, String text) throws SQLException {
        execute("INSERT INTO briefs(title,date,text) VALUES (?,?,?)",
                title, date, text);
        return getLastInsertId();
    }

    public List<Facility> getFacilities() throws SQLException {
        List<Facility> result = new ArrayList<>();

        try (Statement s = connection.createStatement();
             ResultSet r = s.executeQuery("SELECT * FROM facilities ORDER BY id")) {
            while (r.next())
                result.add(new Facility(r.getInt("id"),
                        r.getString("name"), r.getString("abbreviation")));
        }
        return result;
    }

    public List<Agent> getAgents() throws SQLException {
        List<Agent> result = new ArrayList<>();

        try (Statement s = connection.createStatement();
             ResultSet r = s.executeQuery("SELECT * FROM agents ORDER BY id")) {
            while (r.next())
                result.add(new Agent(r.getInt("id"), r.getString("name"),
                        r.getString("date_of_birth"), r.getString("date_of_death"),
                        r.getString("notes"), r.getInt("facility_id")));
        }
        return result;
    }

    public List<Brief> getBriefs() throws SQLException {
        List<Brief> result = new ArrayList<>();

        try (Statement s = connection.createStatement();
             ResultSet r = s.executeQuery("SELECT * FROM briefs ORDER BY id")) {
            while (r.next())
                result.add(new Brief(r.getInt("id"), r.getString("title"),
                        r.getString("date"), r.getString("text")));
        }
        return result;
    }

    public void assignMission(int missionId, int facilityId,
                              List<Integer> agentIds) throws SQLException {
        if (agentIds == null || agentIds.isEmpty())
            throw new IllegalArgumentException("At least one agent required");

        for (Integer id : agentIds)
            if (id == null)
                throw new IllegalArgumentException("Agent ID cannot be null");

        if (!connection.getAutoCommit())
            throw new SQLException("Connection must use auto-commit");

        connection.setAutoCommit(false);
        try {
            execute("INSERT INTO mission_facilities(mission_id,facility_id) VALUES (?,?)",
                    missionId, facilityId);

            for (int id : new LinkedHashSet<>(agentIds))
                execute("INSERT INTO mission_agents(mission_id,agent_id) VALUES (?,?)",
                        missionId, id);

            connection.commit();
        } catch (SQLException | RuntimeException e) {
            connection.rollback();
            throw e;
        } finally {
            connection.setAutoCommit(true);
        }
    }

    public boolean updateBrief(int id, String title,
                               String date, String text) throws SQLException {
        return execute("UPDATE briefs SET title=?, date=?, text=? WHERE id=?",
                title, date, text, id) > 0;
    }

    public boolean updateAgent(int id, String name, String dob,
                               String dod, String notes, int facilityId) throws SQLException {
        return execute("UPDATE agents SET name=?, date_of_birth=?, " +
                        "date_of_death=?, notes=?, facility_id=? WHERE id=?",
                name, dob, dod, notes, facilityId, id) > 0;
    }

    public boolean updateFacility(int id, String name,
                                  String abbreviation) throws SQLException {
        return execute("UPDATE facilities SET name=?, abbreviation=? WHERE id=?",
                name, abbreviation, id) > 0;
    }
}

    public String decipherFile(String fileName) {
        return decipherFile(fileName, "key.txt");
    }

    // Get and decipher a file using a key file from the ciphers folder
    public String decipherFile(String fileName, String keyFileName) {
        String contents = fileHandler.readDataFile(fileName);
        String key = fileHandler.readCipherFile(keyFileName);

        if (contents == null || key == null) {
            return null;
        }

        try {
            Cipher cipher = new Cipher(key);
            return cipher.decipher(contents);
        } catch (IllegalArgumentException e) {
            return null; // bad key: the UI shows an error instead of crashing
        }
    }

// Audit log stored in the SQLite database (Member D).
public class SqliteAuditLog implements AuditLog {

    private final String url;
    private final Clock clock;

    public SqliteAuditLog(Path dbFile) {
        this(dbFile, Clock.systemDefaultZone());
    }

    // The clock can be fixed in tests so timestamps are predictable
    public SqliteAuditLog(Path dbFile, Clock clock) {
        Path file = dbFile.toAbsolutePath();
        this.url = "jdbc:sqlite:" + file;
        this.clock = clock;
        try {
            Files.createDirectories(file.getParent());
            try (Connection conn = DriverManager.getConnection(url);
                 Statement st = conn.createStatement()) {
                st.execute("CREATE TABLE IF NOT EXISTS audit_log ("
                        + "id INTEGER PRIMARY KEY AUTOINCREMENT, "
                        + "timestamp TEXT NOT NULL, "
                        + "username TEXT NOT NULL, "
                        + "action TEXT NOT NULL, "
                        + "item TEXT NOT NULL)");
            }
        } catch (IOException | SQLException e) {
            throw new IllegalStateException("Cannot set up the audit log", e);
        }
    }

    @Override
    public void record(String username, String action, String item) {
        if (isBlank(username) || isBlank(action) || isBlank(item)) {
            throw new IllegalArgumentException("Audit events need a user, an action, and an item");
        }
        String time = LocalDateTime.now(clock).truncatedTo(ChronoUnit.SECONDS).toString();
        String sql = "INSERT INTO audit_log (timestamp, username, action, item) VALUES (?, ?, ?, ?)";
        try (Connection conn = DriverManager.getConnection(url);
             PreparedStatement ps = conn.prepareStatement(sql)) {
            ps.setString(1, time);
            ps.setString(2, username);
            ps.setString(3, action);
            ps.setString(4, item);
            ps.executeUpdate();
        } catch (SQLException e) {
            throw new IllegalStateException("Cannot write to the audit log", e);
        }
    }

    @Override
    public List<AuditEntry> getEntries() {
        List<AuditEntry> entries = new ArrayList<>();
        String sql = "SELECT id, timestamp, username, action, item FROM audit_log ORDER BY id";
        try (Connection conn = DriverManager.getConnection(url);
             Statement st = conn.createStatement();
             ResultSet rs = st.executeQuery(sql)) {
            while (rs.next()) {
                entries.add(new AuditEntry(rs.getInt("id"), rs.getString("timestamp"),
                        rs.getString("username"), rs.getString("action"), rs.getString("item")));
            }
        } catch (SQLException e) {
            throw new IllegalStateException("Cannot read the audit log", e);
        }
        return entries;
    }

    private static boolean isBlank(String s) {
        return s == null || s.isBlank();
    }
}

/**
 * SQLite implementation of MissionStore.
 * This is the ONLY class in the project that uses java.sql.
 */
public class SqliteMissionStore implements MissionStore {

    // The database lives in database/ (not data/ or ciphers/)
    public static final Path DEFAULT_DB_FILE = Path.of("database", "missions.db");
    public static final Path DEFAULT_TSV_FILE = Path.of("database", "mission_briefs.tsv");

    private static final String SELECT = "SELECT id, title, brief, date FROM missions";

    private final String url;

    // creates the table if needed and loads the TSV. Safe to run every time.
    public SqliteMissionStore() {
        this(DEFAULT_DB_FILE);
        if (Files.exists(DEFAULT_TSV_FILE)) {
            importFromTsv(DEFAULT_TSV_FILE);
        }
    }

    // Uses the given database file (created if missing) and creates the table if needed.
    public SqliteMissionStore(Path dbFile) {
        Path file = dbFile.toAbsolutePath();
        url = "jdbc:sqlite:" + file;
        try {
            Files.createDirectories(file.getParent());
            try (Connection conn = DriverManager.getConnection(url);
                 Statement st = conn.createStatement()) {
                // UNIQUE (title, date) stops the same mission being added twice
                st.execute("CREATE TABLE IF NOT EXISTS missions ("
                        + "id INTEGER PRIMARY KEY AUTOINCREMENT, "
                        + "title TEXT NOT NULL, "
                        + "brief TEXT NOT NULL, "
                        + "date TEXT NOT NULL, "
                        + "UNIQUE (title, date))");
            }
        } catch (IOException | SQLException e) {
            throw new IllegalStateException("Cannot set up the mission database", e);
        }
    }

    @Override
    public List<Mission> getAllMissions() {
        return query(SELECT + " ORDER BY id", null);
    }

    @Override
    public Mission getMission(int id) {
        List<Mission> found = query(SELECT + " WHERE id = ?", id);
        return found.isEmpty() ? null : found.get(0);
    }

    @Override
    public List<Mission> searchBriefs(String term) {
        if (term == null || term.isBlank()) {
            return new ArrayList<>();
        }
        // instr() is a plain "contains" test, so % and _ are treated as normal characters
        return query(SELECT + " WHERE instr(LOWER(brief), LOWER(?)) > 0 ORDER BY id", term.trim());
    }

    @Override
    public int importFromTsv(Path tsvFile) {
        int added = 0;
        String sql = "INSERT OR IGNORE INTO missions (title, brief, date) VALUES (?, ?, ?)";
        try (Connection conn = DriverManager.getConnection(url)) {
            conn.setAutoCommit(false); // one save at the end is much faster
            try (PreparedStatement ps = conn.prepareStatement(sql)) {
                for (Mission m : readTsv(tsvFile)) {
                    ps.setString(1, m.getTitle());
                    ps.setString(2, m.getBrief());
                    ps.setString(3, m.getDate());
                    added += ps.executeUpdate(); // 1 = added, 0 = already there
                }
            }
            conn.commit();
        } catch (SQLException e) {
            throw new IllegalStateException("Cannot import missions", e);
        }
        return added;
    }

    // Runs a SELECT (with at most one ? parameter) and turns every row into a Mission.
    private List<Mission> query(String sql, Object parameter) {
        List<Mission> result = new ArrayList<>();
        try (Connection conn = DriverManager.getConnection(url);
             PreparedStatement ps = conn.prepareStatement(sql)) {
            if (parameter != null) {
                ps.setObject(1, parameter);
            }
            try (ResultSet rs = ps.executeQuery()) {
                while (rs.next()) {
                    result.add(new Mission(rs.getInt("id"), rs.getString("title"),
                            rs.getString("brief"), rs.getString("date")));
                }
            }
        } catch (SQLException e) {
            throw new IllegalStateException("Cannot read missions", e);
        }
        return result;
    }

    // Reads Title Date Text lines with tabs between. Skips the header, blank lines and bad lines.
    static List<Mission> readTsv(Path file) {
        List<Mission> missions = new ArrayList<>();
        try {
            for (String line : Files.readAllLines(file, StandardCharsets.UTF_8)) {
                String[] parts = line.split("\t", -1);
                if (parts.length != 3 || line.contains("Title\tDate")) {
                    continue; // header, blank line, or wrong number of columns
                }
                String title = parts[0].trim();
                String date = parts[1].trim();
                String brief = parts[2].trim();
                if (!title.isEmpty() && !date.isEmpty() && !brief.isEmpty()) {
                    missions.add(new Mission(0, title, brief, date)); // database assigns the id
                }
            }
        } catch (IOException e) {
            throw new IllegalStateException("Cannot read import file: " + file, e);
        }
        return missions;
    }

import java.util.List;

public class UserInterface {

    private final ProgramControl programControl;

    public UserInterface(ProgramControl programControl) {
        this.programControl = programControl;
    }

    public void run(String[] array) {
        if (array.length == 0) {
            displayFileList();
        } else {
            displayFileContents(array);
        }
    }

    private void displayFileList() {
        List<String> files = programControl.getAvailableFiles();
        for (int i = 0; i < files.size(); i++) {
            String number = (i + 1) < 10 ? "0" + (i + 1) : "" + (i + 1);
            System.out.println(number + " " + files.get(i));
        }
    }

    private void displayFileContents(String[] array) {
        int fileNumber;
        try {
            fileNumber = Integer.parseInt(array[0]);
        } catch (NumberFormatException e) {
            System.out.println("Error: '" + array[0] + "' is not a valid file number.");
            return;
        }

        List<String> files = programControl.getAvailableFiles();
        if (fileNumber < 1 || fileNumber > files.size()) {
            System.out.println("Error: No file found matching number " + fileNumber + ".");
            return;
        }
        String fileName = files.get(fileNumber - 1);

        String contents;
        if (array.length > 1) {
            contents = programControl.decipherFile(fileName, array[1]);
        } else {
            contents = programControl.decipherFile(fileName);
        }

        if (contents == null) {
            System.out.println("Error: Unable to read or decipher '" + fileName + "'.");
            return;
        }

        System.out.println(contents);
    }
